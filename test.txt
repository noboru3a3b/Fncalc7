primes = make_vector(100, 0);
primes[0] = 2;
primes[1] = 3;
index = 2;

def is_prime(n)
  let i= 1, p = 0 in
    while (i < index) do
      p = primes[i];
      if n < (p * p) then return 1; end
      if modulo(n, p) == 0 then return -1; end
      i = i + 1;
    end
    1;
  end
end

def calc_primes(n)
  let i = 5 in
    while (i < n) do
      if is_prime(i) > 0 then
        primes[index] = i;
        index = index + 1;
        if index >= 100 then return 0; end
      end
      i = i + 2;
    end
  end
end

calc_primes(500);

primes;
